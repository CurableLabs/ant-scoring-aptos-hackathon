Lab Credit NFT (Lane 1) Code:
// --- CureDAO Lab Credit NFT  (Lane 1)---
// This module defines the "Lab Credit Badge", a non-transferable NFT (soulbound token)
// that acts as a verifiable receipt for an inventor's IP submission.
// It is intentionally simple, focusing on provenance and ownership.
// Built for the Aptos Ctrl+MOVE Hackathon.
// By: The Dream Team
// ---
module cure_dao_admin::lab_credit {
   use std::signer;
   use std::string::{Self, String};
   use aptos_framework::object::{Self, Object, ConstructorRef};
   use aptos_token_objects::token::{Self, Token};

   // === Errors ===
   /// The module has not been initialized.
   const E_NOT_INITIALIZED: u64 = 1;
   /// The caller is not the protocol contract.
   const E_NOT_PROTOCOL: u64 = 2;

   // === Structs ===

   /// The capability to manage the Lab Credit collection. Stored under the deployer account.
   struct LabCreditManager has key {
       collection: Object<Collection>,
       mint_ref: ConstructorRef,
   }

   /// The main Lab Credit NFT. This will be an object owned by the inventor.
   struct LabCredit has key {
       // The hash of the intellectual property submission.
       ip_hash: String,
       // The original creator of the IP.
       creator_address: address,
   }

   /// The collection metadata for all Lab Credits.
   struct Collection has key {}


   // === Public Functions ===

   /// Initializes the Lab Credit collection. This must be called by the deployer account once.
   public entry fun initialize(deployer: &signer) {
       let deployer_addr = signer::address_of(deployer);
       assert!(deployer_addr == @cure_dao_admin, E_NOT_INITIALIZED); // Restrict to deployer

       let collection_constructor = object::create_named_object(deployer, b"CurableDAO Lab Credit Collection");
       let collection_signer = object::generate_signer(&collection_constructor);

       let collection = token::create_collection(
           &collection_signer,
           string::utf8(b"CurableDAO Lab Credits"), // name
           string::utf8(b"Non-transferable badges for IP contributors."), // description
           string::utf8(b"https://curable.dao/lab_credit.png") // url
       );

       let mint_ref = object::generate_constructor_ref(&collection_constructor);

       move_to(deployer, LabCreditManager {
           collection,
           mint_ref
       });
   }

   /// Mints a new Lab Credit Badge.
   /// This is a private function callable only by our `protocol` contract to ensure
   /// badges are only created as part of the official IP registration flow.
   public(friend) fun submit_ip(
       protocol_signer: &signer,
       creator: address,
       ip_hash: String,
       submission_id: u64
   ) -> Object<Token> acquires LabCreditManager {
       assert!(signer::address_of(protocol_signer) == @cure_dao_admin, E_NOT_PROTOCOL);
      
       let manager = borrow_global<LabCreditManager>(@cure_dao_admin);
       let token_constructor_ref = object::generate_constructor_ref(&object::create_object_from_account(protocol_signer));

       // Create the token. Note that we are NOT enabling royalty or transferability features.
       // The absence of a public `transfer` function makes this token effectively soulbound.
       let lab_credit_token = token::create_named_token(
           &token_constructor_ref,
           manager.collection,
           string::utf8(b"Lab Credit"),
           string::utf8(b""), // description
           string::format(&string::utf8(b"Lab Credit #{}"), &vector[submission_id]), // name
           string::utf8(b"https://curable.dao/lab_credit_badge.svg") // url
       );

       // Add the IP data to the token itself
       move_to(
           object::signer_from_constructor_ref(&token_constructor_ref),
           LabCredit {
               ip_hash,
               creator_address: creator,
           }
       );

       // Directly transfer the newly minted token to the creator.
       object::transfer(lab_credit_token, creator);
       lab_credit_token
   }
}


?Next is the pd_cure.move contract. This defines the PDCure fungible token, the utility token for community validation in Lane 2, including the fixed-price sale logic.

PDCure Utility Token (Lane 2)
// --- CureDAO PDCure Utility Token (Lane 2) ---
// This module defines the PDCure token, a fungible utility token used for
// community validation, staking, and governance within the CurableDAO ecosystem.
// It includes logic for a fixed-price sale during Phase 1.
// Built for the Aptos Ctrl+MOVE Hackathon.
// ---
module cure_dao_admin::pd_cure {
   use std::signer;
   use aptos_framework::coin::{Self, Coin, MintCapability, BurnCapability};
   use aptos_framework::aptos_coin::AptosCoin;

   // === Errors ===
   const E_MODULE_NOT_INITIALIZED: u64 = 1;
   const E_INSUFFICIENT_APT: u64 = 2;

   // === Structs ===

   /// The metadata for the PDCure token.
   struct PDCure has store { }

   /// Capabilities to manage the PDCure token supply. Stored under the deployer account.
   struct PDCureManager has key {
       mint_cap: MintCapability<PDCure>,
       burn_cap: BurnCapability<PDCure>,
   }

   // === Constants ===
  
   /// The fixed exchange rate: 1 APT = 100 PDCure.
   const PDCURE_PER_APT: u64 = 100;

   // === Public Functions ===

   /// Initializes the PDCure token. Must be called by the deployer account once.
   public entry fun initialize(deployer: &signer) {
       let (burn_cap, mint_cap) = coin::initialize<PDCure>(
           deployer,
           b"PDCure Token",
           b"PDCURE",
           8, // decimals
           true, // monitor supply
       );

       move_to(deployer, PDCureManager {
           mint_cap,
           burn_cap
       });
   }

   /// Allows a user to buy PDCure tokens at a fixed rate using AptosCoin.
   /// This simulates the Phase 1 fixed-price sale.
   public entry fun buy_fixed_price(buyer: &signer, apt_amount: u64) acquires PDCureManager {
       let buyer_addr = signer::address_of(buyer);
       let manager = borrow_global<PDCureManager>(@cure_dao_admin);

       // Ensure user has a PDCure coin store registered.
       if (!coin::is_account_registered<PDCure>(buyer_addr)) {
           coin::register<PDCure>(buyer);
       };

       // Calculate the amount of PDCure to mint.
       let pdcure_to_mint = apt_amount * PDCURE_PER_APT;

       // Withdraw APT from the buyer and transfer to the DAO treasury.
       let apt_coins = coin::withdraw<AptosCoin>(buyer, apt_amount);
       coin::deposit(@cure_dao_admin, apt_coins);

       // Mint the new PDCure tokens and deposit them into the buyer's account.
       let pdcure_coins = coin::mint<PDCure>(pdcure_to_mint, &manager.mint_cap);
       coin::deposit(buyer_addr, pdcure_coins);
   }

   /// Friend function to allow the protocol to burn staked tokens if needed.
   public(friend) fun burn_staked(amount: u64, burn_cap: &BurnCapability<PDCure>) {
       let coins_to_burn = coin::mint<PDCure>(amount, &coin::generate_mint_capability(burn_cap));
       coin::burn(coins_to_burn, burn_cap);
   }
}				

?Finally, the protocol.move contract. This is the brain of the operation, connecting Lane 1 and Lane 2. It manages IP submissions, handles community staking, and triggers the critical "phase transition" from Validating to Validated.

CureDAO Main Protocol (The Brain)
// --- CureDAO Main Protocol ---
// This module is the central nervous system of the CureDAO platform.
// It manages the lifecycle of IP submissions, connecting the inventor's Lab Credit (Lane 1)
// with the community's validation staking using PDCure tokens (Lane 2).
// Built for the Aptos Ctrl+MOVE Hackathon.
// ---
module cure_dao_admin::protocol {
   use std::signer;
   use std::string::{Self, String};
   use aptos_framework::object::{Self, Object};
   use aptos_framework::coin::{Self, Coin, BurnCapability};
   use aptos_std::table::{Self, Table};
  
   use cure_dao_admin::lab_credit;
   use cure_dao_admin::pd_cure::{Self, PDCure};

   // === Errors ===
   const E_PROTOCOL_NOT_INITIALIZED: u64 = 1;
   const E_IP_NOT_FOUND: u64 = 2;
   const E_ALREADY_VALIDATED: u64 = 3;

   // === Structs ===

   /// Represents the status of an IP submission.
   struct IPStatus has store, drop, copy {
       is_submitted: bool,
       is_validating: bool,
       is_validated: bool,
   }

   /// Stores all data related to a single IP submission.
   struct IPSubmission has store {
       id: u64,
       creator: address,
       ip_hash: String,
       status: IPStatus,
       staked_amount: u64,
       lab_credit_nft: Object<aptos_token_objects::token::Token>,
       // A table to store who staked how much. Key: staker_address, Value: amount_staked
       stakers: Table<address, u64>,
   }

   /// The main resource holding the state of the entire protocol.
   struct ProtocolState has key {
       // A table of all IP submissions. Key: submission_id, Value: IPSubmission
       ip_submissions: Table<u64, IPSubmission>,
       next_submission_id: u64,
       burn_cap: BurnCapability<PDCure>,
   }

   // === Constants ===
  
   /// The amount of staked PDCure required to validate an IP.
   const VALIDATION_THRESHOLD: u64 = 1000000; // e.g., 1,000,000 PDCure

   // === Public Functions ===

   /// Initializes the protocol. Must be called by the deployer once.
   public entry fun initialize(deployer: &signer) {
       let burn_cap = coin::create_burn_capability<PDCure>(deployer);

       move_to(deployer, ProtocolState {
           ip_submissions: table::new(),
           next_submission_id: 1,
           burn_cap,
       });
   }

   /// An inventor calls this to register their IP.
   /// This function creates the Lab Credit NFT (Lane 1) and lists the IP for validation (Lane 2).
   public entry fun register_ip(creator: &signer, ip_hash: String) acquires ProtocolState {
       let state = borrow_global_mut<ProtocolState>(@cure_dao_admin);
       let creator_addr = signer::address_of(creator);
       let submission_id = state.next_submission_id;

       // Mint the Lab Credit Badge via a friend call.
       let lab_credit_token = lab_credit::submit_ip(
           object::create_signer(@cure_dao_admin),
           creator_addr,
           ip_hash,
           submission_id
       );

       // Create the new submission record.
       let new_submission = IPSubmission {
           id: submission_id,
           creator: creator_addr,
           ip_hash: string::utf8(b""), // Re-assigning to avoid copy issues
           status: IPStatus { is_submitted: true, is_validating: true, is_validated: false },
           staked_amount: 0,
           lab_credit_nft: lab_credit_token,
           stakers: table::new(),
       };
       // Update hash after creation
       new_submission.ip_hash = ip_hash;

       table::add(&mut state.ip_submissions, submission_id, new_submission);
       state.next_submission_id = submission_id + 1;
   }

   /// A community member stakes their PDCure to vote for an IP's validation.
   public entry fun stake_for_validation(staker: &signer, submission_id: u64, amount: u64) acquires ProtocolState {
       let state = borrow_global_mut<ProtocolState>(@cure_dao_admin);
       assert!(table::contains(&state.ip_submissions, submission_id), E_IP_NOT_FOUND);

       let submission = table::borrow_mut(&mut state.ip_submissions, submission_id);
       assert!(!submission.status.is_validated, E_ALREADY_VALIDATED);

       // Withdraw PDCure from the staker.
       let pdcure_coins = coin::withdraw<PDCure>(staker, amount);
      
       // For the hackathon, we will burn the staked tokens to keep the logic simple.
       // A real-world scenario might lock them in a treasury.
       coin::burn_from(signer::address_of(staker), pdcure_coins, &state.burn_cap);

       submission.staked_amount = submission.staked_amount + amount;

       // Check if this stake pushes the submission over the validation threshold.
       if (submission.staked_amount >= VALIDATION_THRESHOLD) {
           submission.status.is_validating = false;
           submission.status.is_validated = true;
       }
   }
}

       let (burn_cap, mint_cap) = coin::initialize<PDCure>(
           deployer,
           b"PDCure Token",
           b"PDCURE",
           8, // decimals
           true, // monitor supply
       );

       move_to(deployer, PDCureManager {
           mint_cap,
           burn_cap
       });
   }

   /// Allows a user to buy PDCure tokens at a fixed rate using AptosCoin.
   /// This simulates the Phase 1 fixed-price sale.
   public entry fun buy_fixed_price(buyer: &signer, apt_amount: u64) acquires PDCureManager {
       let buyer_addr = signer::address_of(buyer);
       let manager = borrow_global<PDCureManager>(@cure_dao_admin);

       // Ensure user has a PDCure coin store registered.
       if (!coin::is_account_registered<PDCure>(buyer_addr)) {
           coin::register<PDCure>(buyer);
       };

       // Calculate the amount of PDCure to mint.
       let pdcure_to_mint = apt_amount * PDCURE_PER_APT;

       // Withdraw APT from the buyer and transfer to the DAO treasury.
       let apt_coins = coin::withdraw<AptosCoin>(buyer, apt_amount);
       coin::deposit(@cure_dao_admin, apt_coins);

       // Mint the new PDCure tokens and deposit them into the buyer's account.
       let pdcure_coins = coin::mint<PDCure>(pdcure_to_mint, &manager.mint_cap);
       coin::deposit(buyer_addr, pdcure_coins);
   }

   /// Friend function to allow the protocol to burn staked tokens if needed.
   public(friend) fun burn_staked(amount: u64, burn_cap: &BurnCapability<PDCure>) {
       let coins_to_burn = coin::mint<PDCure>(amount, &coin::generate_mint_capability(burn_cap));
       coin::burn(coins_to_burn, burn_cap);
   }
}

// --- CureDAO Main Protocol ---
// This module is the central nervous system of the CureDAO platform.
// It manages the lifecycle of IP submissions, connecting the inventor's Lab Credit (Lane 1)
// with the community's validation staking using PDCure tokens (Lane 2).
// Built for the Aptos Ctrl+MOVE Hackathon.
// ---
module cure_dao_admin::protocol {
   use std::signer;
   use std::string::{Self, String};
   use aptos_framework::object::{Self, Object};
   use aptos_framework::coin::{Self, Coin, BurnCapability};
   use aptos_std::table::{Self, Table};
  
   use cure_dao_admin::lab_credit;
   use cure_dao_admin::pd_cure::{Self, PDCure};

   // === Errors ===
   const E_PROTOCOL_NOT_INITIALIZED: u64 = 1;
   const E_IP_NOT_FOUND: u64 = 2;
   const E_ALREADY_VALIDATED: u64 = 3;

   // === Structs ===

   /// Represents the status of an IP submission.
   struct IPStatus has store, drop, copy {
       is_submitted: bool,
       is_validating: bool,
       is_validated: bool,
   }

   /// Stores all data related to a single IP submission.
   struct IPSubmission has store {
       id: u64,
       creator: address,
       ip_hash: String,
       status: IPStatus,
       staked_amount: u64,
       lab_credit_nft: Object<aptos_token_objects::token::Token>,
       // A table to store who staked how much. Key: staker_address, Value: amount_staked
       stakers: Table<address, u64>,
   }

   /// The main resource holding the state of the entire protocol.
   struct ProtocolState has key {
       // A table of all IP submissions. Key: submission_id, Value: IPSubmission
       ip_submissions: Table<u64, IPSubmission>,
       next_submission_id: u64,
       burn_cap: BurnCapability<PDCure>,
   }

   // === Constants ===
  
   /// The amount of staked PDCure required to validate an IP.
   const VALIDATION_THRESHOLD: u64 = 1000000; // e.g., 1,000,000 PDCure

   // === Public Functions ===

   /// Initializes the protocol. Must be called by the deployer once.
   public entry fun initialize(deployer: &signer) {
       let burn_cap = coin::create_burn_capability<PDCure>(deployer);

       move_to(deployer, ProtocolState {
           ip_submissions: table::new(),
           next_submission_id: 1,
           burn_cap,
       });
   }

   /// An inventor calls this to register their IP.
   /// This function creates the Lab Credit NFT (Lane 1) and lists the IP for validation (Lane 2).
   public entry fun register_ip(creator: &signer, ip_hash: String) acquires ProtocolState {
       let state = borrow_global_mut<ProtocolState>(@cure_dao_admin);
       let creator_addr = signer::address_of(creator);
       let submission_id = state.next_submission_id;

       // Mint the Lab Credit Badge via a friend call.
       let lab_credit_token = lab_credit::submit_ip(
           object::create_signer(@cure_dao_admin),
           creator_addr,
           ip_hash,
           submission_id
       );

       // Create the new submission record.
       let new_submission = IPSubmission {
           id: submission_id,
           creator: creator_addr,
           ip_hash: string::utf8(b""), // Re-assigning to avoid copy issues
           status: IPStatus { is_submitted: true, is_validating: true, is_validated: false },
           staked_amount: 0,
           lab_credit_nft: lab_credit_token,
           stakers: table::new(),
       };
       // Update hash after creation
       new_submission.ip_hash = ip_hash;

       table::add(&mut state.ip_submissions, submission_id, new_submission);
       state.next_submission_id = submission_id + 1;
   }

   /// A community member stakes their PDCure to vote for an IP's validation.
   public entry fun stake_for_validation(staker: &signer, submission_id: u64, amount: u64) acquires ProtocolState {
       let state = borrow_global_mut<ProtocolState>(@cure_dao_admin);
       assert!(table::contains(&state.ip_submissions, submission_id), E_IP_NOT_FOUND);

       let submission = table::borrow_mut(&mut state.ip_submissions, submission_id);
       assert!(!submission.status.is_validated, E_ALREADY_VALIDATED);

       // Withdraw PDCure from the staker.
       let pdcure_coins = coin::withdraw<PDCure>(staker, amount);
      
       // For the hackathon, we will burn the staked tokens to keep the logic simple.
       // A real-world scenario might lock them in a treasury.
       coin::burn_from(signer::address_of(staker), pdcure_coins, &state.burn_cap);

       submission.staked_amount = submission.staked_amount + amount;

       // Check if this stake pushes the submission over the validation threshold.
       if (submission.staked_amount >= VALIDATION_THRESHOLD) {
           submission.status.is_validating = false;
           submission.status.is_validated = true;
       }
   }
}
